State Management: Sealed ViewState<T>


// 1. Định nghĩa sealed state exhaustive
sealed class ViewState<T> {
  const ViewState();
}
final class ViewLoading<T> extends ViewState<T> { const ViewLoading(); }
final class ViewEmpty<T> extends ViewState<T> { const ViewEmpty(); }
final class ViewFailure<T> extends ViewState<T> {
  final String message;
  const ViewFailure(this.message);
}
final class ViewSuccess<T> extends ViewState<T> {
  final T data;
  const ViewSuccess(this.data);
}

// 2. Switch expression an toàn trong build()
Widget build(BuildContext context, WidgetRef ref) {
  final state = ref.watch(todayVictoryViewModelProvider);
  return switch (state) {
    ViewLoading() => const ShimmerVictoryCard(),
    ViewEmpty() => const EmptyJarMascot(),
    ViewFailure(:final message) => ErrorNotice(message: message),
    ViewSuccess(:final data) => VictoryCard(note: data),
  };
}
